cmake_minimum_required(VERSION 3.25)
project(StellarBladeSupportNatives LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "MSVC x64 is required")
endif()
set(SBCORE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(SBCORE_BUILD_SHIM_PROBE OFF CACHE BOOL "" FORCE)
set(SBCORE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/sbcore")
add_subdirectory(sbcore)
include("${SBCORE_DIR}/cmake/sbcore_preset.cmake")
function(native target)
 add_library(${target} SHARED ${ARGN})
 sbcore_apply_native_preset(${target})
 target_link_libraries(${target} PRIVATE sbcore::sbcore sbcore::ue4ss)
endfunction()
native(SBGodNative god/dllmain.cpp god/god_hook.cpp god/god_identity.cpp god/god_sites.cpp god/god_gate.cpp god/god_state.cpp)
target_include_directories(SBGodNative PRIVATE god)
native(SBLiveAddNative liveadd/dllmain.cpp liveadd/live_add.cpp)
target_include_directories(SBLiveAddNative PRIVATE liveadd)
target_link_libraries(SBLiveAddNative PRIVATE bcrypt)
native(SBMovementNative movement/src/dllmain.cpp movement/src/movement.cpp)
target_include_directories(SBMovementNative PRIVATE movement/src)
# Retry Point imports seven symbols, so generate its exact dedicated import library.
set(RETRY_LIB "${CMAKE_BINARY_DIR}/lib/retry_ue4ss.lib")
add_custom_command(OUTPUT "${RETRY_LIB}"
 COMMAND "${CMAKE_AR}" /nologo /Brepro /machine:x64 "/def:${CMAKE_CURRENT_SOURCE_DIR}/retry/ue4ss/UE4SS.def" "/out:${RETRY_LIB}"
 DEPENDS retry/ue4ss/UE4SS.def VERBATIM)
add_custom_target(retry_imports DEPENDS "${RETRY_LIB}")
add_library(SBRetryPointNative SHARED retry/dllmain.cpp retry/retry_point.cpp)
sbcore_apply_native_preset(SBRetryPointNative)
target_include_directories(SBRetryPointNative PRIVATE retry)
target_link_libraries(SBRetryPointNative PRIVATE sbcore::sbcore "${RETRY_LIB}")
add_dependencies(SBRetryPointNative retry_imports)
